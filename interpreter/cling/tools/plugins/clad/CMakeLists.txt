include(ExternalProject)

set(clad_install_dir ${CMAKE_BINARY_DIR}/etc/cling/)
# Specify include dirs for clad
set(CLAD_INCLUDE_DIRS ${clad_install_dir})

#-------------------------------------------------------------------------------
# There are two ways to build clad, and both end up providing the cladPlugin and
# cladDifferentiator targets plus the headers in etc/cling/plugins/include:
#
#  * clad_in_tree=OFF (the default): clad is an external project with its own
#    CMake invocation, built from a download or from CLAD_SOURCE_DIR.
#
#  * clad_in_tree=ON: the sources in CLAD_SOURCE_DIR become part of the ROOT
#    project. Touching clad then triggers an incremental ROOT build, both are
#    compiled with one and the same set of flags, and clad shows up in the
#    compilation database - which is what it takes to profile and optimise ROOT
#    and clad as a whole.
#-------------------------------------------------------------------------------

if(clad_in_tree)

  # With builtin_llvm, LLVM defines these in its own directory scope only, so
  # they do not reach us here the way they do for an installed LLVM.
  if(NOT LLVM_VERSION_MAJOR)
    set(LLVM_VERSION_MAJOR ${ROOT_LLVM_VERSION_REQUIRED_MAJOR})
  endif()
  if(NOT LLVM_CMAKE_DIR)
    set(LLVM_CMAKE_DIR ${CMAKE_SOURCE_DIR}/interpreter/llvm-project/llvm/cmake/modules)
  endif()

  #---Options read by clad's CMakeLists.txt------------------------------------
  # Take LLVM and Clang as this build already found them, instead of letting
  # clad look for them on its own.
  set(CLAD_BUILT_AS_SUBPROJECT ON)
  # ROOT pastes the static libraries into libCling, it has no use for clad.so.
  # This also switches off clad's own test and benchmark suites, which need a
  # clang binary to run and are not ours to build.
  set(CLAD_BUILD_STATIC_ONLY ON)

  add_subdirectory(${CLAD_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR}/clad EXCLUDE_FROM_ALL)

  # With builtin_llvm/builtin_clang, the clang headers clad includes are
  # generated by the build itself; make sure they are there first. An external
  # clang installation brings them along, and defines no such target.
  if(TARGET clangBasic)
    add_dependencies(cladDifferentiator clangBasic)
    add_dependencies(cladPlugin clangBasic)
  endif()

  #---Stage the clad headers where the interpreter looks for them--------------
  # This reproduces the layout of the external project's install step (see the
  # -I in TCling.cxx and the install() in the top-level CMakeLists.txt), so
  # that nothing downstream can tell the two ways of building clad apart.
  set(clad_staged_include ${clad_install_dir}/plugins/include)
  set(clad_staged_files)

  macro(CLAD_STAGE_FILE source destination)
    add_custom_command(OUTPUT ${destination}
      COMMAND ${CMAKE_COMMAND} -E copy_if_different ${source} ${destination}
      DEPENDS ${source}
      VERBATIM)
    list(APPEND clad_staged_files ${destination})
  endmacro()

  file(GLOB_RECURSE clad_headers CONFIGURE_DEPENDS
    RELATIVE ${CLAD_SOURCE_DIR}/include
    ${CLAD_SOURCE_DIR}/include/*.h ${CLAD_SOURCE_DIR}/include/*.def)
  foreach(header ${clad_headers})
    CLAD_STAGE_FILE(${CLAD_SOURCE_DIR}/include/${header} ${clad_staged_include}/${header})
  endforeach()

  file(GLOB clad_tool_headers CONFIGURE_DEPENDS
    RELATIVE ${CLAD_SOURCE_DIR}/tools ${CLAD_SOURCE_DIR}/tools/*.h)
  foreach(header ${clad_tool_headers})
    CLAD_STAGE_FILE(${CLAD_SOURCE_DIR}/tools/${header} ${clad_staged_include}/clad/tools/${header})
  endforeach()

  # Configured by clad's lib/Differentiator/CMakeLists.txt.
  CLAD_STAGE_FILE(${CMAKE_CURRENT_BINARY_DIR}/clad/include/clad/Differentiator/Version.inc
                  ${clad_staged_include}/clad/Differentiator/Version.inc)

  CLAD_STAGE_FILE(${CLAD_SOURCE_DIR}/cmake/modules/AddClad.cmake
                  ${clad_install_dir}/plugins/lib/cmake/clad/AddClad.cmake)

  add_custom_target(clad-headers DEPENDS ${clad_staged_files})
  add_dependencies(cladDifferentiator clad-headers)

  return()
endif()

# Clad Libraries
set(CLAD_CXX_FLAGS ${CMAKE_CXX_FLAGS})
set(_clad_build_type ${CMAKE_CFG_INTDIR})
if(MSVC AND NOT CMAKE_GENERATOR MATCHES Ninja)
  if (winrtdebug)
    set(_clad_build_type Debug)
  else()
    set(_clad_build_type Release)
  endif()
  set(EXTRA_BUILD_ARGS --config ${_clad_build_type})
endif()
if(NOT _clad_build_type STREQUAL "" AND NOT _clad_build_type STREQUAL ".")
  set(EXTRA_BUILD_ARGS --config ${_clad_build_type})
endif()

if(NOT MSVC AND CMAKE_VERSION VERSION_LESS 3.31.1)
  # Workaround for https://gitlab.kitware.com/cmake/cmake/-/issues/26398
  # The problem got eventually fixed upstream in CMake.
  list(APPEND EXTRA_BUILD_ARGS "-j 1")
endif()

set(_CLAD_LIBRARY_PATH ${CMAKE_CURRENT_BINARY_DIR}/clad-prefix/src/clad-build/${_clad_build_type}/lib${LLVM_LIBDIR_SUFFIX})

# build byproducts only needed by Ninja
if(CMAKE_GENERATOR MATCHES Ninja)
  set(CLAD_BYPRODUCTS
    ${_CLAD_LIBRARY_PATH}/${CMAKE_STATIC_LIBRARY_PREFIX}cladPlugin${CMAKE_STATIC_LIBRARY_SUFFIX}
    ${_CLAD_LIBRARY_PATH}/${CMAKE_STATIC_LIBRARY_PREFIX}cladDifferentiator${CMAKE_STATIC_LIBRARY_SUFFIX}
  )
endif()

if(APPLE)
  set(_clad_extra_cmake_args -DCMAKE_OSX_SYSROOT=${CMAKE_OSX_SYSROOT})
endif()

if (CMAKE_CXX_STANDARD)
  list(APPEND _clad_extra_cmake_args -DCMAKE_CXX_STANDARD=${CMAKE_CXX_STANDARD})
endif(CMAKE_CXX_STANDARD)

if (LLVM_FORCE_USE_OLD_TOOLCHAIN)
  list(APPEND _clad_extra_cmake_args -DLLVM_FORCE_USE_OLD_TOOLCHAIN=${LLVM_FORCE_USE_OLD_TOOLCHAIN})
endif(LLVM_FORCE_USE_OLD_TOOLCHAIN)

list(APPEND _clad_extra_cmake_args -DCLAD_BUILD_STATIC_ONLY=ON)

# Wrap download, configure and build steps in a script to log output
set(_clad_extra_settings
  LOG_DOWNLOAD ON
  LOG_CONFIGURE ON
  LOG_BUILD ON
  LOG_INSTALL ON
  LOG_OUTPUT_ON_FAILURE ON
)

# If the CLAD_SOURCE_DIR variable is defined in the CMake configuration, we're
# skipping the download of the repository and use the passed directory.
if (DEFINED CLAD_SOURCE_DIR)
  list(APPEND _clad_extra_settings SOURCE_DIR ${CLAD_SOURCE_DIR})
else()
  list(APPEND _clad_extra_settings GIT_REPOSITORY https://github.com/vgvassilev/clad.git)
  list(APPEND _clad_extra_settings GIT_TAG v2.4)
endif()

# list(APPEND _clad_patches_list "patch1.patch" "patch2.patch")
list(APPEND _clad_patches_list "nullptr_dereference_fix.patch" "windows_build_fix.patch")
set(_clad_patch_command
      ${CMAKE_COMMAND} -E copy_directory
      ${CMAKE_SOURCE_DIR}/interpreter/cling/tools/plugins/clad/patches <SOURCE_DIR>
      && git checkout -q <SOURCE_DIR>
      && git apply --ignore-space-change --ignore-whitespace ${_clad_patches_list}
      )

ExternalProject_Add(
  clad
  UPDATE_COMMAND ""
  PATCH_COMMAND ${_clad_patch_command}
  CMAKE_ARGS -G ${CMAKE_GENERATOR}
             -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}
             -DCMAKE_C_COMPILER=${CMAKE_C_COMPILER}
             -DCMAKE_C_FLAGS=${CMAKE_C_FLAGS}
             -DCMAKE_CXX_COMPILER=${CMAKE_CXX_COMPILER}
             -DCMAKE_CXX_FLAGS=${CLAD_CXX_FLAGS}
             -DCMAKE_CXX_FLAGS_DEBUG=${CMAKE_CXX_FLAGS_DEBUG}
             -DCMAKE_INSTALL_PREFIX=${clad_install_dir}/plugins
             -DLLVM_DIR=${LLVM_DIR}
             -DClang_DIR=${CLANG_CMAKE_DIR}
             ${_clad_extra_cmake_args}
  BUILD_COMMAND ${CMAKE_COMMAND} --build . ${EXTRA_BUILD_ARGS}
  INSTALL_COMMAND ${CMAKE_COMMAND} --build . ${EXTRA_BUILD_ARGS} --target install
  BUILD_BYPRODUCTS ${CLAD_BYPRODUCTS}
  ${_clad_extra_settings}
  # We need the target clangBasic to be built before building clad. However, we
  # support building prebuilt clang and adding clangBasic breaks this case.
  # Delegate the dependency resolution to the clingInterpreter target (which
  # will always depend on clangBasic).
  DEPENDS clingInterpreter
  )

# Register cladPlugin, cladDifferentiator
foreach (lib cladPlugin cladDifferentiator)
  add_library(${lib} IMPORTED STATIC GLOBAL)
  add_dependencies(${lib} clad)
endforeach()

set_property(TARGET cladPlugin PROPERTY IMPORTED_LOCATION ${_CLAD_LIBRARY_PATH}/${CMAKE_STATIC_LIBRARY_PREFIX}cladPlugin${CMAKE_STATIC_LIBRARY_SUFFIX})
set_property(TARGET cladDifferentiator PROPERTY IMPORTED_LOCATION ${_CLAD_LIBRARY_PATH}/${CMAKE_STATIC_LIBRARY_PREFIX}cladDifferentiator${CMAKE_STATIC_LIBRARY_SUFFIX})
